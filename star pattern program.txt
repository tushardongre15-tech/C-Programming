#include <stdio.h>

void row(int spaces, int stars) {
    while (spaces--) putchar(' ');
    while (stars--) putchar('*');
    putchar('\n');
}

int main() {
    int n = 4, i;

    for (i = 1; i <= n; i++)     row(3 * n - i, 2 * i - 1);
    for (i = 1; i <= n; i++)     row(i - 1, 6 * n - 2 * i + 1);
    for (i = n - 1; i >= 1; i--) row(i - 1, 6 * n - 2 * i + 1);
    for (i = n; i >= 1; i--)     row(3 * n - i, 2 * i - 1);

    return 0;
}


#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main() {
    int user, computer;

    srand(time(NULL));

    printf("1. Rock\n");
    printf("2. Paper\n");
    printf("3. Scissors\n");
    printf("Enter your choice: ");
    scanf("%d", &user);

    if (user < 1 || user > 3) {
        printf("Invalid choice!");
        return 0;
    }

    computer = rand() % 3 + 1;

    printf("Computer chose: %d\n", computer);

    if (user == computer)
        printf("It's a draw!");
    else if ((user == 1 && computer == 3) ||
             (user == 2 && computer == 1) ||
             (user == 3 && computer == 2))
        printf("You win!");
    else
        printf("Computer wins!");

    return 0;
}
